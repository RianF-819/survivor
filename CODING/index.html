Required Supabase SQL Setup: 
CREATE TABLE IF NOT EXISTS highscores (
  id BIGSERIAL PRIMARY KEY,
  nama TEXT NOT NULL,
  skor_tertinggi INTEGER NOT NULL,
  room_code TEXT NOT NULL DEFAULT 'GLOBAL',
